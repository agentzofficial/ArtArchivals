<script>
(() => {
  const overlay = document.getElementById('membership-modal-overlay');

  function openMembershipModal(video) {
    document.getElementById('membership-modal-image').src = video.image;
    document.getElementById('membership-modal-title').textContent = video.title;
    document.getElementById('membership-modal-date').textContent = formatDate(video.date);
    document.getElementById('membership-modal-description').textContent = video.description;

    overlay.classList.remove('hidden');
  }

  function formatDate(dateStr) {
    const d = new Date(dateStr);
    return d.toLocaleDateString('en-GB', {
      day: 'numeric',
      month: 'long',
      year: 'numeric'
    });
  }

  function closeMembershipModal() {
    overlay.classList.add('hidden');
  }

  fetch('/assets/data/memberships.json')
    .then(r => r.json())
    .then(data => {

      const purpleGrid = document.getElementById('membership-purple-video-grid');
      const blueGrid   = document.getElementById('membership-blue-video-grid');

      data.videos
        .sort((a,b) => new Date(b.date) - new Date(a.date))
        .forEach(video => {
          const img = document.createElement('img');
          img.src = video.image;
          img.alt = video.title;
          img.addEventListener('click', () => openMembershipModal(video));

          if (video.tier === 'purple') purpleGrid.appendChild(img);
          if (video.tier === 'blue')   blueGrid.appendChild(img);
        });
    });

  overlay.addEventListener('click', e => {
    if (e.target === overlay) closeMembershipModal();
  });

  document
    .querySelector('.membership-modal-close')
    .addEventListener('click', closeMembershipModal);

})();
(() => {
  const container = document.getElementById('membership-posts-container');

  function formatDate(dateStr) {
    return new Date(dateStr).toLocaleDateString('en-GB', {
      day: 'numeric',
      month: 'long',
      year: 'numeric'
    });
  }
  fetch('/assets/data/memberships.json')
    .then(r => r.json())
    .then(data => {
      const postsData = data.posts;
      const months = Object.keys(postsData).sort((a,b) => new Date(b+'-01') - new Date(a+'-01'));

      months.forEach(month => {
        const monthDiv = document.createElement('div');
        monthDiv.className = 'membership-posts-month';
        const monthLabel = new Date(month+'-01').toLocaleDateString('en-GB', { month: 'long', year: 'numeric' });
        monthDiv.textContent = monthLabel;

        const itemsDiv = document.createElement('div');
        itemsDiv.className = 'membership-posts-items';
        itemsDiv.style.display = 'none';

        postsData[month].forEach(post => {
          const p = document.createElement('p');
          p.textContent = `${post.type} / ${formatDate(post.date)}`;
          itemsDiv.appendChild(p);
        });

        monthDiv.addEventListener('click', () => {
          monthDiv.classList.toggle('open');
          itemsDiv.style.display = itemsDiv.style.display === 'block' ? 'none' : 'block';
        });

        container.appendChild(monthDiv);
        container.appendChild(itemsDiv);
      });
    });
})();
</script>

<script>
(() => {
  function reorderTierContent() {
    const tierContents = document.querySelectorAll('.tier-content');

    if (window.innerWidth <= 768) {
      tierContents.forEach(tc => {
        const img = tc.querySelector('img');
        const text = tc.querySelector('.tier-text');
        if (img && text) {
          tc.innerHTML = '';
          tc.appendChild(img);
          tc.appendChild(text);
        }
      });
    } else {
      tierContents.forEach(tc => {
        const img = tc.querySelector('img');
        const text = tc.querySelector('.tier-text');
        if (!img || !text) return;
        if (tc.classList.contains('image-left')) {
          tc.innerHTML = '';
          tc.appendChild(img);
          tc.appendChild(text);
        } else if (tc.classList.contains('image-right')) {
          tc.innerHTML = '';
          tc.appendChild(text);
          tc.appendChild(img);
        }
      });
    }
  }

  window.addEventListener('resize', reorderTierContent);
  window.addEventListener('load', reorderTierContent);
})();
</script>
